// Dimensions of the grid
export const MATRIX_WIDTH = 10;
export const MATRIX_HEIGHT = 20;
// Create an empty matrix (grid)
export function createMatrix() {
return Array.from({ length: MATRIX_HEIGHT }, () =>
Array(MATRIX_WIDTH).fill(0)
);
}
// Merge the current piece into the matrix
export function merge(matrix, piece) {
const { shape, pos } = piece;
const newMatrix = matrix.map(row => [...row]);
shape.forEach((row, y) => {
row.forEach((value, x) => {
if (value !== 0) {
newMatrix[y + pos.y][x + pos.x] = value;
}
});
});
return newMatrix;
}
// Check for collision
export function checkCollision(matrix, piece) {
const { shape, pos } = piece;
for (let y = 0; y < shape.length; y++) {
for (let x = 0; x < shape[y].length; x++) {
if (
shape[y][x] !== 0 &&
(matrix[y + pos.y] &&
matrix[y + pos.y][x + pos.x]) !== 0
) {
return true;
}
}
}
return false;
}
// Clear completed lines and return new matrix + number cleared
export function clearLines(matrix) {
const newMatrix = matrix.filter(row => row.some(cell => cell === 0));
const linesCleared = MATRIX_HEIGHT - newMatrix.length;
while (newMatrix.length < MATRIX_HEIGHT) {
newMatrix.unshift(Array(MATRIX_WIDTH).fill(0));
}
return { newMatrix, linesCleared };
}
