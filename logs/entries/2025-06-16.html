import { useEffect, useRef } from 'react';

export function useGameLoop({ actors = [], isPaused = false, tickRate = 50 }) {
const loopRef = useRef(null);

useEffect(() => {
if (isPaused || actors.length === 0) return;

loopRef.current = setInterval(() => {
actors.forEach(actor => {
if (typeof actor.tick === 'function') {
actor.tick(tickRate);
}
});
}, tickRate);

return () => clearInterval(loopRef.current);
}, [actors, isPaused, tickRate]);

export function useGameLoop({ actors = [], isPaused = false, tickRate = 50 }) {
const loopRef = useRef(null);
const [, forceRender] = useState(0); // used to trigger React rerender when needed

useEffect(() => {
if (isPaused || actors.length === 0) return;

actors.forEach(actor => actor.clearDirty?.());
}
}, tickRate);

return () => clearInterval(loopRef.current);

// hooks/useGameLoop.js
import { useEffect, useRef, useState } from 'react';

export function useGameLoop({ actors = [], isPaused = false, tickRate = 50 }) {
const loopRef = useRef(null);
const [, forceRender] = useState(0); // used to trigger React rerender when needed

useEffect(() => {
if (isPaused || actors.length === 0) return;

loopRef.current = setInterval(() => {
actors.forEach(actor => {
if (typeof actor.tick === 'function') {
actor.tick(tickRate);
}
});

const shouldRender = actors.some(actor => actor.shouldRender?.());
if (shouldRender) {
forceRender(c => c + 1);
actors.forEach(actor => actor.clearDirty?.());
}
}, tickRate);

return () => clearInterval(loopRef.current);
}, [actors, isPaused, tickRate]);
}
