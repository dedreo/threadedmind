import React, { useState, useEffect, useRef } from 'react';
import WymC from './components/WymC';

export default function App() {
const [messages, setMessages] = useState([

]);

const [input, setInput] = useState("");
const [prePrompt, setPrePrompt] = useState("You are WymC, an assistant AI");
const [lastUsedPrompt, setLastUsedPrompt] = useState("You are WymC, an assistant AI");
const [logEvents, setLogEvents] = useState([]);
const [debugEnabled, setDebugEnabled] = useState(true);
const [contextActive, setContextActive] = useState(true);
const [loading, setLoading] = useState(false);
const messageEndRef = useRef(null);

useEffect(() => {
messageEndRef.current?.scrollIntoView({ behavior: 'smooth' });
}, [messages, loading]);

const appendLog = (actor, message) => {
if (!debugEnabled) return;
const timestamp = new Date().toLocaleTimeString();
setLogEvents(prev => [...prev.slice(-19), `[${timestamp}] [${actor}] ${message}`]);
};

const handleSubmit = async () => {
if (!input.trim()) return;

const userMessage = { sender: 'You', text: input.trim() };
setMessages(prev => [...prev, userMessage]);
appendLog("You", `Submitted input: \"${input.trim()}\"`);
setInput("");
setLoading(true);

try {
const fullPrompt = `${prePrompt}\nUser: ${input.trim()}\nWymC:`;
const res = await fetch('/api/wymc', {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify({ prompt: fullPrompt })
});
const data = await res.json();
const botMessage = { sender: 'WymC', text: data.reply || "..." };
setMessages(prev => [...prev, botMessage]);
setLastUsedPrompt(prePrompt);
appendLog("WymC", `Generated response (${data.reply.length} chars)`);
} catch (err) {
setMessages(prev => [...prev, { sender: 'WymC', text: "System failure. Figures." }]);
appendLog("WymC", "Backend error during generation");
} finally {
setLoading(false);
}
};

const handleKeyDown = (e) => {
if (e.key === 'Enter' && !e.shiftKey) {
e.preventDefault();
handleSubmit();
}
};

const handlePrePromptChange = (e) => {
setPrePrompt(e.target.value);
// Temporarily suppress log noise
// appendLog("You", "Modified pre-prompt");
};

const flushContext = () => {
setContextActive(false);
appendLog("You", "Flushed session context");
};

const isPrePromptDirty = prePrompt !== lastUsedPrompt;

return (
<div className="h-screen overflow-hidden bg-black text-[#00ff44] font-mono p-4 grid grid-cols-[1fr_auto] gap-4 relative">

<span className={msg.sender === 'You' ? 'italic text-[#33ffaa]' : ''}> {msg.text}</span>
</div>
))}
{loading && (
<div className="border border-green-400 p-2 animate-pulse">
<strong>WymC:</strong> ...
</div>
)}
<div ref={messageEndRef} />
</div>

<textarea
rows={2}
placeholder="Type your message..."
value={input}
onChange={(e) => setInput(e.target.value)}
onKeyDown={handleKeyDown}
className="resize-none border border-green-400 bg-[#0a0a0a] text-[#00ff44] p-2 w-full caret-[#00ff44] placeholder-green-600"
autoFocus
/>
</form>

{/* Right side: Debug / Context panel */}

<label className="text-xs flex items-center gap-1">
<input type="checkbox" checked={debugEnabled} onChange={() => setDebugEnabled(!debugEnabled)} /> Logging
</label>
<button
onClick={flushContext}
className="border border-green-400 text-[#00ff44] px-2 py-1 text-xs bg-black hover:bg-[#003300]"
>
Flush Context
</button>
</div>

{/* 3 stacked boxes */}
<div className="col-span-3 flex flex-col gap-2">
<div className={`border border-green-400 p-2 ${!contextActive ? 'opacity-40' : ''}`}>
<label className="block text-xs mb-1">Current pre-prompt</label>
<textarea
rows={5}
value={prePrompt}
onChange={handlePrePromptChange}
className={`w-full bg-[#0a0a0a] ${isPrePromptDirty ? 'text-yellow-400' : 'text-[#00ff44]'} border border-green-400 p-1 resize-none caret-[#00ff44]`}
/>
</div>

<div className={`border border-green-400 p-2 overflow-y-auto max-h-32 text-xs whitespace-pre-line ${!contextActive ? 'opacity-40' : ''}`}>
{logEvents.length > 0 ? logEvents.map((log, idx) => (
<div key={idx}>{log}</div>
)) : <div className="italic text-[#339966]">No system events yet</div>}
</div>

<div className={`border border-green-400 p-2 ${!contextActive ? 'opacity-40' : ''}`}>Files or lists updates?</div>
</div>
</div>

{/* Floating avatar top-right */}
<div className="absolute top-4 right-4 z-10">

alt="WymC holding pumpkin"

/>
</div>
</div>
);

### 🧠 Now about: **Logging** and **Flush Context**
You're right to question how those behave. Let’s unpack each:

---

### ✅ `Logging` (the checkbox)
**What it *does***:
When `debugEnabled` is `true`, the `appendLog()` function records log lines like:
```
[11:29:47 AM] [You] Submitted input: "Hey what's up?"
[11:29:48 AM] [WymC] Generated response (165 chars)
```
They get stored in the `logEvents` array and rendered in the log box.

**What it *doesn’t* do**:
- It does *not* track system errors beyond input/output.
- It does *not* persist across reloads.
- It does *not* change behavior of the AI call itself—just whether those events show up in the debug box.

So toggling it off stops those logs from accumulating—but the AI still works the same.

---

### 🧹 `Flush Context` (the button)
**What it *actually does right now***:
- Sets `contextActive = false`
- Lowers opacity of the debug/pre-prompt boxes (`opacity-40`)

**But crucially…**
- It **does not** reset the `messages` array.
- It **does not** reset the pre-prompt.
- It **does not** tell the AI to forget anything—unless you manually change the prePrompt or rebuild the system.
