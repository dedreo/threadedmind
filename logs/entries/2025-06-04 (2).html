other => other.id !== blob.id && other.x === blob.x && other.y === belowY
);

if (isBlobBelow) {

for (const dx of directions) {
const newX = blob.x + dx;
const spaceAvailable =
newX >= 0 &&
newX < matrix[0].length &&
matrix[blob.y][newX] === 0 &&
!allBlobs.some(b => b.x === newX && b.y === blob.y);

if (spaceAvailable) {
return { ...blob, x: newX }; // sidestep around friend
}
}
return blob; // stuck next to buddy
}

const canFall =
belowY < matrix.length &&
matrix[belowY][blob.x] === 0 &&
!allBlobs.some(
other => other.id !== blob.id && other.x === blob.x && other.y === belowY
);

return canFall ? { ...blob, y: belowY } : blob;
}

export function nudgeBlob(blob, matrix, allBlobs, piece) {
let newX = blob.x;

if (Math.random() < 0.33) {

const testX = blob.x + direction;

const blocked =
testX < 0 ||
testX >= matrix[0].length ||
matrix[blob.y][testX] !== 0 ||
allBlobs.some(other => other.id !== blob.id && other.x === testX && other.y === blob.y) ||
piece.shape.some((row, dy) =>
row.some((val, dx) =>
val !== 0 &&
piece.pos.x + dx === testX &&
piece.pos.y + dy === blob.y
)
);

if (!blocked) {
newX = testX;
}
}

const belowY = blob.y + 1;
const canFall =
belowY < matrix.length &&
matrix[belowY][newX] === 0 &&
!allBlobs.some(other => other.id !== blob.id && other.x === newX && other.y === belowY);

const newY = canFall ? belowY : blob.y;

return { ...blob, x: newX, y: newY };
}
