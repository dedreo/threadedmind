export function determineBlobState(blob, board, playerPos) {

if (blob.squished) return "squished";

export function shouldBlobMove(blob, matrix) {
// Simple AI: 1/3 chance to try moving if not blocked
if (blob.isFrozen || blob.squished) return false;

}

resolved.push(blob);

resolved.push(blob);

idleFrames: 0
};
